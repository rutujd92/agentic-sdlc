# Design

Add a feature endpoint group UrlShortener.Api.Links.QrCodeEndpoints (MapQrCodeEndpoints) mapped in Program.cs after MapLinkEndpoints. Resolve the link via ILinkRepository.GetByCodeAsync, build the short URL from IOptions<ShortenerOptions>.BaseUrl (never the Host header, per CLAUDE.md), and render a PNG with QRCoder's PngByteQRCode (pure managed, no System.Drawing, works in Linux containers). ECC level Q tolerates print damage.

## Impacted files
- src/UrlShortener.Api/UrlShortener.Api.csproj
- src/UrlShortener.Api/Links/QrCodeEndpoints.cs
- src/UrlShortener.Api/Program.cs
- tests/UrlShortener.Tests/Api/QrCodeEndpointTests.cs
- README.md

## API and schema changes
GET /api/links/{code}/qr -> 200 image/png (bytes start with the PNG signature) | 404 application/problem+json. New dependency: QRCoder 1.8.0 (MIT) in UrlShortener.Api. Tests use ShortenerApiFactory and create links via POST /api/links with aliases.

## Risks
- New third-party dependency (license and supply chain): QRCoder is MIT and widely used; requires human sign-off
- Image generation cost per request: small (<1ms); no caching needed at current scale

+using Microsoft.AspNetCore.Http.HttpResults;
+using Microsoft.Extensions.Options;
+using QRCoder;
+using UrlShortener.Core.Links;
+
+namespace UrlShortener.Api.Links;
+
+public static class QrCodeEndpoints
+{
+    /// <summary>Pixels per QR module; 10 yields roughly a 290px image for a short URL.</summary>
+    private const int PixelsPerModule = 10;
+
+    public static IEndpointRouteBuilder MapQrCodeEndpoints(this IEndpointRouteBuilder app)
+    {
+        app.MapGet("/api/links/{code}/qr", GetQrCodeAsync).WithName("GetLinkQrCode");
+        return app;
+    }
+
+    private static async Task<Results<FileContentHttpResult, ProblemHttpResult>> GetQrCodeAsync(
+        string code,
+        ILinkRepository repository,
+        IOptions<ShortenerOptions> options,
+        CancellationToken cancellationToken)
+    {
+        var link = await repository.GetByCodeAsync(code, cancellationToken);
+        if (link is null)
+        {
+            return TypedResults.Problem(statusCode: StatusCodes.Status404NotFound, title: "Link not found", detail: $"No link exists for '{code}'.");
+        }
+
+        var shortUrl = $"{options.Value.BaseUrl.ToString().TrimEnd('/')}/{link.Code}";
+        using var generator = new QRCodeGenerator();
+        using var data = generator.CreateQrCode(shortUrl, QRCodeGenerator.ECCLevel.Q);
+        using var png = new PngByteQRCode(data);
+        return TypedResults.File(png.GetGraphic(PixelsPerModule), "image/png");
+    }
+}

+using System.Net;
+using System.Net.Http.Json;
+
+namespace UrlShortener.Tests.Api;
+
+public sealed class QrCodeEndpointTests : IAsyncLifetime
+{
+    private static readonly byte[] PngSignature = [0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A];
+    private readonly ShortenerApiFactory _factory = new();
+    private HttpClient _client = null!;
+
+    public async Task InitializeAsync()
+    {
+        await _factory.InitializeDatabaseAsync();
+        _client = _factory.CreateClient();
+    }
+
+    public async Task DisposeAsync()
+    {
+        _client.Dispose();
+        await _factory.DisposeAsync();
+    }
+
+    [Fact]
+    public async Task Qr_code_for_existing_link_is_a_png_image()
+    {
+        await _client.PostAsJsonAsync("/api/links", new { url = "https://example.com", alias = "qr-me" });
+
+        var response = await _client.GetAsync("/api/links/qr-me/qr");
+
+        Assert.Equal(HttpStatusCode.OK, response.StatusCode);
+        Assert.Equal("image/png", response.Content.Headers.ContentType?.MediaType);
+        var bytes = await response.Content.ReadAsByteArrayAsync();
+        Assert.True(bytes.Length > 100);
+        Assert.Equal(PngSignature, bytes[..PngSignature.Length]);
+    }
+
+    [Fact]
+    public async Task Qr_code_for_unknown_link_returns_404_problem_details()
+    {
+        var response = await _client.GetAsync("/api/links/missing/qr");
+
+        Assert.Equal(HttpStatusCode.NotFound, response.StatusCode);
+        Assert.Equal("application/problem+json", response.Content.Headers.ContentType?.MediaType);
+    }
+
+    [Fact]
+    public async Task Different_links_produce_different_qr_codes()
+    {
+        await _client.PostAsJsonAsync("/api/links", new { url = "https://example.com/a", alias = "qr-a" });
+        await _client.PostAsJsonAsync("/api/links", new { url = "https://example.com/b", alias = "qr-b" });
+
+        var a = await _client.GetByteArrayAsync("/api/links/qr-a/qr");
+        var b = await _client.GetByteArrayAsync("/api/links/qr-b/qr");
+
+        Assert.NotEqual(a, b);
+    }
+}
